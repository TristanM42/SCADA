Expression :    Sodium_Nitrite_bags_SP
Déclencheur :   Changement de valeur
Bande morte :   0

dim v_total as float; ' reference volume for water + glycol + sodium nitrite
dim v_w as float; ' reference volume for water
dim v_g as float; ' reference volume for glycol
dim v_q as float; ' quantized final volume (= rounded down)
dim v_sn as float; ' reference volume for sodium nitrite
dim v_a as float; ' available volume = max volume that can be filled

v_total = 940; 'InTouch:
' test for cold :
v_w = 558; 'InTouch:
v_g = 287; 'InTouch:

' Calculate v_q from references (csv data for one tank and one recipe)
v_sn = v_total - (v_w+v_g);
v_F_bag = v_sn_per_bag * (1+(v_total-v_sn)/v_sn);
v_q = Sodium_Nitrite_bags_SP * v_F_bag;

' Calculate the volumes of wate, glycol and sodium nitrite
Water_SP = v_q * v_w/v_total;
Glycol_SP = v_q * v_g/v_total;
v_nsn = v_sn_per_bag * Sodium_Nitrite_bags_SP;

v_n = Water_SP + Glycol_SP + v_nsn;
v_a = Tank_Max - (v_n + GA2_PV);
if v_a < 0 then
    Sodium_Nitrite_bags_SP = 0;
    Water_SP = 0;
    Glycol_SP = 0;
    v_nsn = 0;
    v_a = Tank_Max - GA2_PV;

    dim graphicInfo as aaGraphic.GraphicInfo;

    graphicInfo.HasTitleBar = false;

    graphicInfo.CustomProperties = cpValues;
    ShowGraphic( graphicInfo );
endif;
