            Only one line break
                dim Pos = 0;
                dim Token_Left = "";
                dim Token_Right = "";
                
                Pos = StringInString(description_, line_break, 1, 0);
                'Ignore first match, start at second :
                Pos = StringInString(description_, line_break, Pos + 1, 0);
                
                if Pos<>0 then
                    Token_Left = StringLeft(description_, Pos - 1);
                    Token_Right = StringRight(description_, StringLen(description_) - Pos - 1);
                    description_ = Token_Left + " ; " + Token_Right;
                endif;
            For more than one : (nb_line_break)
                dim Pos = 0;
                dim Token_Left = "";
                dim Token_Right = "";
                
                LogMessage("DEBUG DL_Line.remove_line_break : START");
                'LogMessage("DEBUG DL_Line.remove_line_break : call = " + remove_line_break_cpt);
                
                while (remove_line_break_cpt < nb_line_break)
                    Pos = StringInString(description_, line_break, 1, 0);
                    'Pos = StringInString(description_, line_break, Pos + 1, 0);
                
                    if Pos<>0 then
                        Token_Left = StringLeft(description_, Pos - 1);
                        Token_Right = StringRight(description_, StringLen(description_) - Pos - 1);
                        description_ = Token_Left + " ; " + Token_Right;
                    endif;
                
                    remove_line_break_cpt = remove_line_break_cpt + 1;

                endwhile;
