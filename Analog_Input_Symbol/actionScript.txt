dim graphicInfo as aaGraphic.GraphicInfo;

graphicInfo.Identity = "I_"+TagRef;

cpValues[2] = new aaGraphic.CustomPropertyValuePair("Precision", Precision, True);
cpValues[3] = new aaGraphic.CustomPropertyValuePair("Access_Name", Access_Name, true);

if TagRef_2=="" then

    graphicInfo.GraphicName = "Analog_Input_Popup";
    graphicInfo.WindowTitle = "Analog Input Popup: "+TagRef;
else
    cpValues[1] = new aaGraphic.CustomPropertyValuePair("TagRef_1", TagRef, true);

    graphicInfo.WindowTitle = "Analog Input Dual Popup: " + TagRef + "/" + TagRef_2;

endif;

graphicInfo.HasTitleBar = false;
'graphicInfo.HasCloseButton = false;
